//ExSummary: Dieses Tutorial demonstriert, wie man jede Seite einer bestehenden PDF um 180 Grad dreht, indem man die Aspose.Pdf Document-Klasse verwendet.
//ExStepSummary:0: Deklarieren Sie die Quell- und Zieldateipfade für die PDF.
//ExStepSummary:1: Überprüfen Sie, ob die Quelldatei existiert, und brechen Sie mit einer Fehlermeldung ab, falls sie nicht vorhanden ist.
//ExStepSummary:2: Erzeugen Sie ein Document-Objekt für die Quell-PDF innerhalb eines using-Blocks, um Ressourcen zu verwalten.
//ExStepSummary:3: Iterieren Sie über jede Page in pdfDoc.Pages und setzen Sie deren Rotate-Eigenschaft auf Rotation.on180, wodurch jede Seite auf den Kopf gestellt wird.
//ExStepSummary:4: Speichern Sie das modifizierte Document in der Zieldatei, um die Drehungen zu übernehmen.
//ExStepSummary:5: Schreiben Sie eine Bestätigungsnachricht in die Konsole, die anzeigt, dass der Vorgang abgeschlossen ist.
//ExStepImage:3:images/1.png
//ExStep:0-
        const string sourceFile = "source_document.pdf";
        const string destinationFile = "flipped_document.pdf";

//ExStep:1-
        if (!File.Exists(sourceFile))
        {
            Console.Error.WriteLine($"Unable to locate file: {sourceFile}");
            return;
        }

//ExStep:2-
        // Open the PDF within a using statement to ensure resources are released
        using (Document pdfDoc = new Document(sourceFile))
        {

//ExStep:3-
            // Walk through each page (collection is 1‑based) and apply a 180° rotation
            foreach (Page pdfPage in pdfDoc.Pages)
            {
                // Use the correct enum member with the 'on' prefix
                pdfPage.Rotate = Rotation.on180;
            }

//ExStep:4-
            // Write the altered document to disk
            pdfDoc.Save(destinationFile);
        }

//ExStep:5-
        Console.WriteLine($"All pages rotated and saved as '{destinationFile}'.");
//ExEnd
