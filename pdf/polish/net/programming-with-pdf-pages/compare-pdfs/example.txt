//ExSummary: Ten samouczek pokazuje, jak porównać dwa pliki PDF o różnych wymiarach stron, synchronizując rozmiary ich pierwszych stron, konfigurować GraphicalPdfComparer i generować PDF, który podkreśla różnice.
//ExStepSummary:0: Zdefiniuj ścieżki plików dla dwóch źródłowych PDF‑ów oraz wynikowego PDF.
//ExStepSummary:1: Sprawdź, czy oba pliki PDF źródłowe istnieją, i przerwij, jeśli którykolwiek z nich jest nieobecny.
//ExStepSummary:2: Rozpocznij blok try, aby przechwycić błędy podczas przetwarzania.
//ExStepSummary:3: Wczytaj dwa pliki PDF do obiektów Document `docA` i `docB` w ramach instrukcji using.
//ExStepSummary:4: Upewnij się, że każdy PDF zawiera co najmniej jedną stronę przed kontynuacją.
//ExStepSummary:5: Pobierz pierwszą stronę każdego dokumentu do porównania.
//ExStepSummary:6: Skopiuj informacje MediaBox z `firstPage` do `secondPage`, aby strony miały te same wymiary.
//ExStepSummary:7: Utwórz i skonfiguruj `GraphicalPdfComparer` z czerwonym podświetleniem, wysoką rozdzielczością i progiem różnicy 1 %.
//ExStepSummary:8: Uruchom porównanie, wygeneruj PDF z różnicami w `outputPdfPath`, wypisz komunikat o zakończeniu i zamknij bloki using oraz try.
//ExStepSummary:9: Obsłuż ArgumentException, które może wystąpić, jeśli wyrównanie rozmiaru stron nadal jest niewystarczające.
//ExStepSummary:10: Przechwyć wszystkie inne nieoczekiwane wyjątki i zgłoś je.
//ExStep:0-
const string firstPdfPath = "sampleA.pdf";
const string secondPdfPath = "sampleB.pdf";
const string outputPdfPath = "diffResult.pdf";

//ExStep:1-
if (!File.Exists(firstPdfPath) || !File.Exists(secondPdfPath))
{
    Console.Error.WriteLine("Missing one or both source PDF files.");
    return;
}

//ExStep:2-
try
{

//ExStep:3-
using (Document docA = new Document(firstPdfPath))
using (Document docB = new Document(secondPdfPath))
{

//ExStep:4-
if (docA.Pages.Count == 0 || docB.Pages.Count == 0)
{
    Console.Error.WriteLine("At least one document does not contain any pages.");
    return;
}

//ExStep:5-
Page firstPage = docA.Pages[1];
Page secondPage = docB.Pages[1];

//ExStep:6-
secondPage.PageInfo = firstPage.PageInfo;

//ExStep:7-
GraphicalPdfComparer comparer = new GraphicalPdfComparer
{
    Color = Aspose.Pdf.Color.Red,
    Resolution = new Resolution(200),
    Threshold = 1
};

//ExStep:8-
comparer.ComparePagesToPdf(firstPage, secondPage, outputPdfPath);
Console.WriteLine($"Comparison finished. Output saved as '{outputPdfPath}'.");
}
}

//ExStep:9-
catch (ArgumentException ex)
{
    Console.Error.WriteLine($"Argument error during comparison: {ex.Message}");
}

//ExStep:10-
catch (Exception ex)
{
    Console.Error.WriteLine($"Unexpected error: {ex.Message}");
}
//ExEnd
