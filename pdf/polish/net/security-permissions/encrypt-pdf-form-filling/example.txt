//ExSummary: Ten samouczek pokazuje, jak otworzyć istniejący plik PDF za pomocą Aspose.Pdf, ograniczyć uprawnienia tak, aby dozwolone było jedynie wypełnianie formularzy, zaszyfrować go hasłami widza i właściciela przy użyciu AES‑256 oraz zapisać chroniony plik.
//ExStepSummary:0: Zdefiniuj ścieżki plików źródłowego i docelowego oraz hasła widza i właściciela.
//ExStepSummary:1: Sprawdź, czy plik PDF źródłowy istnieje i w razie braku przerwij działanie, wyświetlając komunikat o błędzie.
//ExStepSummary:2: Rozpocznij blok try, aby obsłużyć ewentualne błędy występujące podczas przetwarzania.
//ExStepSummary:3: Otwórz istniejący plik PDF, tworząc obiekt Document o nazwie pdfDoc.
//ExStepSummary:4: Ustaw uprawnienia, aby zezwalały wyłącznie na wypełnianie formularzy, zaszyfruj dokument hasłami widza i właściciela przy użyciu AES‑256 i zapisz chroniony plik PDF.
//ExStepSummary:5: Wypisz komunikat potwierdzający na konsolę i zamknij blok try.
//ExStepSummary:6: Przechwyć wszystkie wyjątki i wyświetl komunikat o błędzie na konsoli.
//ExStep:0-
        const string sourceFile = "sample.pdf";
        const string destFile = "protected.pdf";
        const string viewerPass = "viewerPass";
        const string adminPass = "adminPass";

//ExStep:1-
        if (!File.Exists(sourceFile))
        {
            Console.Error.WriteLine($"Unable to locate file: {sourceFile}");
            return;
        }

//ExStep:2-
        try
        {

//ExStep:3-
            using (Document pdfDoc = new Document(sourceFile))

//ExStep:4-
            {
                Permissions allowed = Permissions.FillForm;
                pdfDoc.Encrypt(viewerPass, adminPass, allowed, CryptoAlgorithm.AESx256);
                pdfDoc.Save(destFile);
            }

//ExStep:5-
            Console.WriteLine($"Secure PDF written to '{destFile}'.");
        }

//ExStep:6-
        catch (Exception ex)
        {
            Console.Error.WriteLine($"Exception occurred: {ex.Message}");
        }
//ExEnd
