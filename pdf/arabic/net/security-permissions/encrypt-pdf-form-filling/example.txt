//ExStepSummary:0: حدد مسارات ملفات المصدر والوجهة وكلمات مرور المشاهد والمالك.
//ExStepSummary:1: تحقق من وجود ملف PDF المصدر وأوقف العملية مع عرض رسالة خطأ إذا لم يكن موجودًا.
//ExStepSummary:2: ابدأ كتلة try لمعالجة أي أخطاء قد تحدث أثناء المعالجة.
//ExStepSummary:3: افتح ملف PDF الموجود بإنشاء كائن Document يُدعى pdfDoc.
//ExStepSummary:4: حدد الأذونات للسماح بملء النماذج فقط، وشفر المستند باستخدام كلمة مرور المشاهد والمالك عبر AES‑256، ثم احفظ ملف PDF المحمي.
//ExStepSummary:5: اكتب رسالة تأكيد إلى وحدة التحكم وأغلق كتلة try.
//ExStepSummary:6: التقط أي استثناءات واعرض رسالة الخطأ في وحدة التحكم.
//ExStep:0-
        const string sourceFile = "sample.pdf";
        const string destFile = "protected.pdf";
        const string viewerPass = "viewerPass";
        const string adminPass = "adminPass";

//ExStep:1-
        if (!File.Exists(sourceFile))
        {
            Console.Error.WriteLine($"Unable to locate file: {sourceFile}");
            return;
        }

//ExStep:2-
        try
        {

//ExStep:3-
            using (Document pdfDoc = new Document(sourceFile))

//ExStep:4-
            {
                Permissions allowed = Permissions.FillForm;
                pdfDoc.Encrypt(viewerPass, adminPass, allowed, CryptoAlgorithm.AESx256);
                pdfDoc.Save(destFile);
            }

//ExStep:5-
            Console.WriteLine($"Secure PDF written to '{destFile}'.");
        }

//ExStep:6-
        catch (Exception ex)
        {
            Console.Error.WriteLine($"Exception occurred: {ex.Message}");
        }
//ExEnd
