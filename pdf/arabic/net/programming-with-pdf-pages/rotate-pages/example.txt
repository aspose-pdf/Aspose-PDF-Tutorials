//ExSummary: يوضح هذا البرنامج التعليمي كيفية تدوير كل صفحة من ملف PDF موجود بمقدار 180 درجة باستخدام فئة Aspose.Pdf Document.
//ExStepSummary:0: حدد مسارات ملفات المصدر والوجهة لملف PDF.
//ExStepSummary:1: تحقق من وجود ملف المصدر وأوقف التنفيذ برسالة خطأ إذا لم يكن موجودًا.
//ExStepSummary:2: أنشئ كائن Document لملف PDF المصدر داخل كتلة using لإدارة الموارد.
//ExStepSummary:3: تجول عبر كل Page في pdfDoc.Pages واضبط خاصية Rotate إلى Rotation.on180، مما يجعل كل صفحة مقلوبة رأسًا على عقب.
//ExStepSummary:4: احفظ الـ Document المعدل إلى ملف الوجهة، مع الحفاظ على عمليات التدوير.
//ExStepSummary:5: اكتب رسالة تأكيد إلى وحدة التحكم تشير إلى إكمال العملية.
//ExStepImage:3:images/1.png
//ExStep:0-
        const string sourceFile = "source_document.pdf";
        const string destinationFile = "flipped_document.pdf";

//ExStep:1-
        if (!File.Exists(sourceFile))
        {
            Console.Error.WriteLine($"Unable to locate file: {sourceFile}");
            return;
        }

//ExStep:2-
        // Open the PDF within a using statement to ensure resources are released
        using (Document pdfDoc = new Document(sourceFile))
        {

//ExStep:3-
            // Walk through each page (collection is 1‑based) and apply a 180° rotation
            foreach (Page pdfPage in pdfDoc.Pages)
            {
                // Use the correct enum member with the 'on' prefix
                pdfPage.Rotate = Rotation.on180;
            }

//ExStep:4-
            // Write the altered document to disk
            pdfDoc.Save(destinationFile);
        }

//ExStep:5-
        Console.WriteLine($"All pages rotated and saved as '{destinationFile}'.");
//ExEnd
