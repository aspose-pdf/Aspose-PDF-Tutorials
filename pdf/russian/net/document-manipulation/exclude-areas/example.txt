//ExSummary: Этот учебник показывает, как использовать Aspose.Pdf для сравнения двух PDF‑файлов бок о бок с игнорированием определённых прямоугольных областей в каждом документе, создавая PDF‑diff, который выделяет только релевантные изменения.
//ExStepSummary:0: Укажите пути к файлам двух исходных PDF и выходному PDF‑diff.
//ExStepSummary:1: Проверьте, что оба исходных PDF‑файла существуют, и при отсутствии любого из них завершите работу с сообщением об ошибке.
//ExStepSummary:2: Создайте массив, содержащий прямоугольную область, которую нужно игнорировать в первом PDF.
//ExStepSummary:3: Создайте массив, содержащий прямоугольную область, которую нужно игнорировать во втором PDF.
//ExStepSummary:4: Настройте сравнение бок о бок, присвоив массивы игнорируемых областей соответствующим свойствам.
//ExStepSummary:5: Загрузите оба PDF и выполните сравнение бок о бок, создав файл diff по указанному пути.
//ExStepSummary:6: Выведите подтверждающее сообщение, указывающее, что сравнение завершено, и где сохранён PDF‑diff.
//ExStep:0-
const string sourcePdfA = "originalA.pdf";
const string sourcePdfB = "originalB.pdf";
const string diffPdfPath = "diff_output.pdf";

//ExStep:1-
if (!File.Exists(sourcePdfA) || !File.Exists(sourcePdfB))
{
    Console.Error.WriteLine("Required PDF files are missing.");
    return;
}

//ExStep:2-
Aspose.Pdf.Rectangle[] zonesToIgnoreA = new Aspose.Pdf.Rectangle[]
{
    new Aspose.Pdf.Rectangle(100, 500, 300, 600) // sample region in the first document
};

//ExStep:3-
Aspose.Pdf.Rectangle[] zonesToIgnoreB = new Aspose.Pdf.Rectangle[]
{
    new Aspose.Pdf.Rectangle(50, 400, 250, 500) // sample region in the second document
};

//ExStep:4-
var comparisonSettings = new SideBySideComparisonOptions
{
    ExcludeAreas1 = zonesToIgnoreA,
    ExcludeAreas2 = zonesToIgnoreB
    // Other settings may be adjusted here, e.g., ExcludeTables = true;
};

//ExStep:5-
using (Document pdfA = new Document(sourcePdfA))
using (Document pdfB = new Document(sourcePdfB))
{
    SideBySidePdfComparer.Compare(pdfA, pdfB, diffPdfPath, comparisonSettings);
}

//ExStep:6-
Console.WriteLine($"Comparison finished. Output written to '{diffPdfPath}'.");
//ExEnd
