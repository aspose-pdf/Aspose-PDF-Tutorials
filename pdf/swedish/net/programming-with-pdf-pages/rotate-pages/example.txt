//ExSummary: Denna handledning visar hur man roterar varje sida i en befintlig PDF med 180 grader med hjälp av Aspose.Pdf Document-klassen.
//ExStepSummary:0: Deklarera käll- och destinationssökvägarna för PDF-filen.
//ExStepSummary:1: Kontrollera att källfilen finns och avbryt med ett felmeddelande om den inte gör det.
//ExStepSummary:2: Skapa ett Document-objekt för käll‑PDF‑filen inom ett using‑block för att hantera resurser.
//ExStepSummary:3: Iterera genom varje Page i pdfDoc.Pages och sätt dess Rotate‑egenskap till Rotation.on180, vilket vänder varje sida upp och ner.
//ExStepSummary:4: Spara det modifierade Document till destinationsfilen och bevara rotationerna.
//ExStepSummary:5: Skriv ett bekräftelsemeddelande till konsolen som indikerar att operationen är slutförd.
//ExStepImage:3:images/1.png
//ExStep:0-
        const string sourceFile = "source_document.pdf";
        const string destinationFile = "flipped_document.pdf";

//ExStep:1-
        if (!File.Exists(sourceFile))
        {
            Console.Error.WriteLine($"Unable to locate file: {sourceFile}");
            return;
        }

//ExStep:2-
        // Open the PDF within a using statement to ensure resources are released
        using (Document pdfDoc = new Document(sourceFile))
        {

//ExStep:3-
            // Walk through each page (collection is 1‑based) and apply a 180° rotation
            foreach (Page pdfPage in pdfDoc.Pages)
            {
                // Use the correct enum member with the 'on' prefix
                pdfPage.Rotate = Rotation.on180;
            }

//ExStep:4-
            // Write the altered document to disk
            pdfDoc.Save(destinationFile);
        }

//ExStep:5-
        Console.WriteLine($"All pages rotated and saved as '{destinationFile}'.");
//ExEnd
