//ExSummary: Den här handledningen visar hur man öppnar en befintlig PDF med Aspose.Pdf, begränsar behörigheter så att endast ifyllning av formulär tillåts, krypterar den med visnings- och ägarlösenord med AES‑256 och sparar den skyddade filen.
//ExStepSummary:0: Definiera sökvägarna för käll- och destinationsfilen samt visnings- och ägarlösenorden.
//ExStepSummary:1: Kontrollera att käll-PDF:en finns och avbryt med ett felmeddelande om den inte gör det.
//ExStepSummary:2: Starta ett try‑block för att hantera eventuella fel som kan uppstå under bearbetningen.
//ExStepSummary:3: Öppna den befintliga PDF-filen genom att skapa ett Document‑objekt med namnet pdfDoc.
//ExStepSummary:4: Ställ in behörigheter så att endast formulärifyllning tillåts, kryptera dokumentet med visnings- och ägarlösenord med AES‑256 och spara den skyddade PDF‑filen.
//ExStepSummary:5: Skriv ett bekräftelsemeddelande till konsolen och avsluta try‑blocket.
//ExStepSummary:6: Fånga eventuella undantag och skriv ut felmeddelandet till konsolen.
//ExStep:0-
        const string sourceFile = "sample.pdf";
        const string destFile = "protected.pdf";
        const string viewerPass = "viewerPass";
        const string adminPass = "adminPass";

//ExStep:1-
        if (!File.Exists(sourceFile))
        {
            Console.Error.WriteLine($"Unable to locate file: {sourceFile}");
            return;
        }

//ExStep:2-
        try
        {

//ExStep:3-
            using (Document pdfDoc = new Document(sourceFile))

//ExStep:4-
            {
                Permissions allowed = Permissions.FillForm;
                pdfDoc.Encrypt(viewerPass, adminPass, allowed, CryptoAlgorithm.AESx256);
                pdfDoc.Save(destFile);
            }

//ExStep:5-
            Console.WriteLine($"Secure PDF written to '{destFile}'.");
        }

//ExStep:6-
        catch (Exception ex)
        {
            Console.Error.WriteLine($"Exception occurred: {ex.Message}");
        }
//ExEnd
