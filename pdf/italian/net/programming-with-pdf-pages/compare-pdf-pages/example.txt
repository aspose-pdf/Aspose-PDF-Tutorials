//ExStart
//ExSummary: Questo tutorial dimostra come utilizzare TextPdfComparer di Aspose.Pdf per eseguire un confronto pagina per pagina di due documenti PDF e generare un PDF di report delle differenze.
//ExStepSummary:0: Dichiarare i percorsi dei file per il PDF originale, il PDF revisionato e il report delle differenze in output.
//ExStepSummary:1: Verificare che entrambi i PDF di origine esistano e interrompere con un messaggio di errore se uno dei due è mancante.
//ExStepSummary:2: Aprire entrambi i PDF, creare le opzioni di confronto predefinite, eseguire un confronto pagina per pagina con `TextPdfComparer.CompareDocumentsPageByPage` e visualizzare un messaggio di successo.
//ExStepSummary:3: Catturare eventuali eccezioni generate durante il confronto e scrivere un messaggio di errore sulla console.
//ExStep:0-
const string sourcePath1 = "original.pdf";
const string sourcePath2 = "revised.pdf";
const string resultPath = "changes_report.pdf";

//ExStep:1-
if (!File.Exists(sourcePath1) || !File.Exists(sourcePath2))
{
    Console.Error.WriteLine("Either of the input PDFs could not be located.");
    return;
}

//ExStep:2-
try
{
    using (Document pdfA = new Document(sourcePath1))
    using (Document pdfB = new Document(sourcePath2))
    {
        ComparisonOptions compareOpts = new ComparisonOptions();
        TextPdfComparer.CompareDocumentsPageByPage(pdfA, pdfB, compareOpts, resultPath);
    }

    Console.WriteLine($"Page‑by‑page diff PDF written to '{resultPath}'.");
}

//ExStep:3-
catch (Exception ex)
{
    Console.Error.WriteLine($"Comparison failed: {ex.Message}");
}
//ExEnd
