//ExSummary: Questo tutorial dimostra come ruotare ogni pagina di un PDF esistente di 180 gradi utilizzando la classe Document di Aspose.Pdf.
//ExStepSummary:0: Dichiarare i percorsi dei file di origine e destinazione per il PDF.
//ExStepSummary:1: Verificare che il file di origine esista e interrompere con un messaggio di errore se non esiste.
//ExStepSummary:2: Creare un oggetto Document per il PDF di origine all'interno di un blocco using per gestire le risorse.
//ExStepSummary:3: Iterare su ogni Page in pdfDoc.Pages e impostare la sua proprietà Rotate su Rotation.on180, capovolgendo ogni pagina.
//ExStepSummary:4: Salvare il Document modificato nel file di destinazione, mantenendo le rotazioni.
//ExStepSummary:5: Scrivere un messaggio di conferma sulla console che indica il completamento dell'operazione.
//ExStepImage:3:images/1.png
//ExStep:0-
        const string sourceFile = "source_document.pdf";
        const string destinationFile = "flipped_document.pdf";

//ExStep:1-
        if (!File.Exists(sourceFile))
        {
            Console.Error.WriteLine($"Unable to locate file: {sourceFile}");
            return;
        }

//ExStep:2-
        // Open the PDF within a using statement to ensure resources are released
        using (Document pdfDoc = new Document(sourceFile))
        {

//ExStep:3-
            // Walk through each page (collection is 1‑based) and apply a 180° rotation
            foreach (Page pdfPage in pdfDoc.Pages)
            {
                // Use the correct enum member with the 'on' prefix
                pdfPage.Rotate = Rotation.on180;
            }

//ExStep:4-
            // Write the altered document to disk
            pdfDoc.Save(destinationFile);
        }

//ExStep:5-
        Console.WriteLine($"All pages rotated and saved as '{destinationFile}'.");
//ExEnd
