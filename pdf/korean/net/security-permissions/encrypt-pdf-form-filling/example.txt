//ExSummary: 이 튜토리얼에서는 Aspose.Pdf를 사용하여 기존 PDF를 열고, 권한을 제한하여 양식 입력만 허용하도록 설정한 다음, AES‑256을 사용해 보기 및 소유자 비밀번호로 암호화하고, 보호된 파일을 저장하는 방법을 보여줍니다.
//ExStepSummary:0: 소스 및 대상 파일 경로와 보기와 소유자 비밀번호를 정의합니다.
//ExStepSummary:1: 소스 PDF가 존재하는지 확인하고, 없을 경우 오류 메시지를 출력하고 중단합니다.
//ExStepSummary:2: 처리 중 발생할 수 있는 오류를 처리하기 위해 try 블록을 시작합니다.
//ExStepSummary:3: Document 객체인 pdfDoc를 생성하여 기존 PDF 파일을 엽니다.
//ExStepSummary:4: 양식 입력만 허용하도록 권한을 설정하고, AES‑256을 사용해 보기와 소유자 비밀번호로 문서를 암호화한 뒤, 보호된 PDF를 저장합니다.
//ExStepSummary:5: 콘솔에 확인 메시지를 출력하고 try 블록을 닫습니다.
//ExStepSummary:6: 예외를 잡아 콘솔에 오류 메시지를 출력합니다.
//ExStep:0-
        const string sourceFile = "sample.pdf";
        const string destFile = "protected.pdf";
        const string viewerPass = "viewerPass";
        const string adminPass = "adminPass";

//ExStep:1-
        if (!File.Exists(sourceFile))
        {
            Console.Error.WriteLine($"Unable to locate file: {sourceFile}");
            return;
        }

//ExStep:2-
        try
        {

//ExStep:3-
            using (Document pdfDoc = new Document(sourceFile))

//ExStep:4-
            {
                Permissions allowed = Permissions.FillForm;
                pdfDoc.Encrypt(viewerPass, adminPass, allowed, CryptoAlgorithm.AESx256);
                pdfDoc.Save(destFile);
            }

//ExStep:5-
            Console.WriteLine($"Secure PDF written to '{destFile}'.");
        }

//ExStep:6-
        catch (Exception ex)
        {
            Console.Error.WriteLine($"Exception occurred: {ex.Message}");
        }
//ExEnd
