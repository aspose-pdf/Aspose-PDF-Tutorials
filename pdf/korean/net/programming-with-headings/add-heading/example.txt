//ExSummary: 이 튜토리얼은 PDF 문서를 생성하고, 기본 언어와 제목을 설정하며, Document, ITaggedContent, StructureElement, HeaderElement 클래스를 사용하여 레벨‑1 헤딩을 삽입하는 방법을 보여줍니다.
//ExStepSummary:0: 생성된 PDF의 출력 파일 이름을 정의합니다.
//ExStepSummary:1: `using` 블록 안에서 새로운 빈 PDF 문서 인스턴스(`pdfDoc`)를 생성합니다.
//ExStepSummary:2: 태그된 PDF 구조를 작업하기 위해 `ITaggedContent` 인터페이스를 얻습니다.
//ExStepSummary:3: 문서의 기본 언어를 English (US)로 설정하고 제목 메타데이터를 할당합니다.
//ExStepSummary:4: 논리 구조 트리의 루트 요소를 가져옵니다.
//ExStepSummary:5: 레벨‑1 헤더 요소를 만들고, 표시 텍스트를 설정하며, 언어를 지정합니다.
//ExStepSummary:6: 헤더 요소를 루트에 추가하여 헤딩이 PDF에 표시되도록 합니다.
//ExStepSummary:7: PDF를 지정된 파일에 저장하고 문서 범위를 닫습니다.
//ExStepSummary:8: 콘솔에 확인 메시지를 출력합니다.
//ExStepImage:6:images/1.png
//ExStep:0-
const string resultFile = "pdf_with_title.pdf";

//ExStep:1-
using (Document pdfDoc = new Document())
{

//ExStep:2-
ITaggedContent tagContent = pdfDoc.TaggedContent;

//ExStep:3-
tagContent.SetLanguage("en-US");
tagContent.SetTitle("PDF Example with Header");

//ExStep:4-
StructureElement rootElement = tagContent.RootElement;

//ExStep:5-
HeaderElement header = tagContent.CreateHeaderElement(1);
header.SetText("Demo Header");
header.Language = "en-US";

//ExStep:6-
rootElement.AppendChild(header);

//ExStep:7-
pdfDoc.Save(resultFile);
}

//ExStep:8-
Console.WriteLine($"PDF generated and stored at '{resultFile}'.");
//ExEnd
