//ExSummary: Este tutorial muestra cómo abrir un PDF existente con Aspose.Pdf, restringir los permisos para que solo se permita rellenar formularios, cifrarlo con contraseñas de visualizador y propietario usando AES‑256 y guardar el archivo protegido.
//ExStepSummary:0: Defina las rutas de archivo de origen y destino, así como las contraseñas de visualizador y propietario.
//ExStepSummary:1: Verifique que el PDF de origen exista y abortar con un mensaje de error si no es así.
//ExStepSummary:2: Inicie un bloque try para manejar cualquier error que pueda ocurrir durante el procesamiento.
//ExStepSummary:3: Abra el archivo PDF existente creando un objeto Document llamado pdfDoc.
//ExStepSummary:4: Establezca los permisos para permitir solo el relleno de formularios, cifre el documento con las contraseñas de visualizador y propietario usando AES‑256 y guarde el PDF protegido.
//ExStepSummary:5: Escriba un mensaje de confirmación en la consola y cierre el bloque try.
//ExStepSummary:6: Capture cualquier excepción y muestre el mensaje de error en la consola.
//ExStep:0-
        const string sourceFile = "sample.pdf";
        const string destFile = "protected.pdf";
        const string viewerPass = "viewerPass";
        const string adminPass = "adminPass";

//ExStep:1-
        if (!File.Exists(sourceFile))
        {
            Console.Error.WriteLine($"Unable to locate file: {sourceFile}");
            return;
        }

//ExStep:2-
        try
        {

//ExStep:3-
            using (Document pdfDoc = new Document(sourceFile))

//ExStep:4-
            {
                Permissions allowed = Permissions.FillForm;
                pdfDoc.Encrypt(viewerPass, adminPass, allowed, CryptoAlgorithm.AESx256);
                pdfDoc.Save(destFile);
            }

//ExStep:5-
            Console.WriteLine($"Secure PDF written to '{destFile}'.");
        }

//ExStep:6-
        catch (Exception ex)
        {
            Console.Error.WriteLine($"Exception occurred: {ex.Message}");
        }
//ExEnd
