//ExSummary: Este tutorial muestra cómo agregar un campo de formulario de cuadro de texto con marcador de posición a un documento PDF y etiquetarlo para accesibilidad usando la biblioteca Aspose.Pdf for .NET.
//ExStepSummary:0: Defina las rutas de los archivos de entrada y salida y verifique que el PDF de origen exista.
//ExStepSummary:1: Abra el archivo PDF existente y cree un objeto Document para trabajar con él.
//ExStepSummary:2: Inserte un TextBoxField en la primera página, establezca su texto de marcador de posición y agréguelo a la colección de formularios.
//ExStepSummary:3: Cree un elemento de estructura lógica /Form, adjúntelo al árbol de contenido etiquetado y asócielo con el campo de cuadro de texto.
//ExStepSummary:4: Guarde el PDF modificado en el archivo de salida especificado y cierre el documento.
//ExStepSummary:5: Escriba un mensaje de confirmación en la consola indicando dónde se guardó el nuevo PDF.
//ExStep:0-
const string sourcePath = "sample_input.pdf";
const string resultPath = "sample_output.pdf";

if (!File.Exists(sourcePath))
{
    Console.Error.WriteLine($"File not found: {sourcePath}");
    return;
}

//ExStep:1-
using (Document pdfDocument = new Document(sourcePath))
{

//ExStep:2-
Page firstPage = pdfDocument.Pages[1];
Aspose.Pdf.Rectangle boxRect = new Aspose.Pdf.Rectangle(100, 600, 300, 630);
TextBoxField placeholderField = new TextBoxField(pdfDocument, boxRect)
{
    Value = "Type your content here..."
};
pdfDocument.Form.Add(placeholderField);

//ExStep:3-
ITaggedContent taggedContent = pdfDocument.TaggedContent;
StructureElement structureRoot = taggedContent.RootElement;
FormElement logicalFormElement = taggedContent.CreateFormElement();
structureRoot.AppendChild(logicalFormElement);
logicalFormElement.Tag(placeholderField);

//ExStep:4-
pdfDocument.Save(resultPath);
}

//ExStep:5-
Console.WriteLine($"PDF with form field saved to '{resultPath}'.");
//ExEnd
