//ExSummary: Este tutorial muestra cómo abrir un PDF existente con Aspose.Pdf, crear un hipervínculo externo etiquetado que incluya texto visible y un título de información emergente, insertarlo en la estructura lógica del documento y guardar el archivo modificado.
//ExStepSummary:0: Defina las rutas del PDF de origen y del archivo de resultado.
//ExStepSummary:1: Verifique que el PDF de origen exista y abortar si no se encuentra.
//ExStepSummary:2: Abra el documento PDF dentro de un bloque using para garantizar su correcta liberación.
//ExStepSummary:3: Obtenga el administrador de contenido etiquetado para el documento abierto.
//ExStepSummary:4: Establezca el idioma del documento a English (US) y dé al PDF un título derivado del nombre del archivo.
//ExStepSummary:5: Recupere el elemento raíz del árbol de estructura lógica al que se agregarán los nuevos elementos.
//ExStepSummary:6: Cree un elemento de enlace, establezca su texto visible, la URL de destino y el título del tooltip, y luego insértelo en la estructura del documento.
//ExStepSummary:7: Guarde el PDF actualizado en el archivo de resultado especificado.
//ExStepSummary:8: Muestre un mensaje de confirmación que indique dónde se guardó el PDF modificado.
//ExStep:0-
const string sourceFile = "source.pdf";
const string resultFile = "result_with_external_link.pdf";

//ExStep:1-
if (!File.Exists(sourceFile))
{
    Console.Error.WriteLine($"Unable to locate file: {sourceFile}");
    return;
}

//ExStep:2-
using (Document pdfDoc = new Document(sourceFile))
{

//ExStep:3-
    ITaggedContent taggedContent = pdfDoc.TaggedContent;

//ExStep:4-
    taggedContent.SetLanguage("en-US");
    taggedContent.SetTitle(Path.GetFileNameWithoutExtension(sourceFile));

//ExStep:5-
    StructureElement rootElement = taggedContent.RootElement;

//ExStep:6-
    LinkElement externalLink = taggedContent.CreateLinkElement();
    externalLink.SetText("Open Aspose PDF library");
    externalLink.Hyperlink = new WebHyperlink("https://www.example.com/pdf");
    externalLink.Title = "Example PDF product page";
    rootElement.AppendChild(externalLink);

//ExStep:7-
    pdfDoc.Save(resultFile);
}

//ExStep:8-
Console.WriteLine($"Modified PDF saved as '{resultFile}'.");
//ExEnd
