//ExStart
//ExSummary: Deze tutorial laat zien hoe je een placeholder-tekstvakformulierveld toevoegt aan een PDF-document en het tagt voor toegankelijkheid met behulp van de Aspose.Pdf for .NET bibliotheek.
//ExStepSummary:0: Definieer de invoer- en uitvoerbestandenpaden en controleer of de bron‑PDF bestaat.
//ExStepSummary:1: Open het bestaande PDF‑bestand en maak een Document‑object aan om mee te werken.
//ExStepSummary:2: Voeg een TextBoxField toe op de eerste pagina, stel de placeholder‑tekst in en voeg het toe aan de formulier‑collectie.
//ExStepSummary:3: Maak een logisch /Form‑structuurelement, koppel het aan de getagde content‑boom en associeer het met het tekstvak‑veld.
//ExStepSummary:4: Sla de gewijzigde PDF op naar het opgegeven uitvoerbestand en sluit het document.
//ExStepSummary:5: Schrijf een bevestigingsbericht naar de console dat aangeeft waar de nieuwe PDF is opgeslagen.
//ExStep:0-
const string sourcePath = "sample_input.pdf";
const string resultPath = "sample_output.pdf";

if (!File.Exists(sourcePath))
{
    Console.Error.WriteLine($"File not found: {sourcePath}");
    return;
}

//ExStep:1-
using (Document pdfDocument = new Document(sourcePath))
{

//ExStep:2-
Page firstPage = pdfDocument.Pages[1];
Aspose.Pdf.Rectangle boxRect = new Aspose.Pdf.Rectangle(100, 600, 300, 630);
TextBoxField placeholderField = new TextBoxField(pdfDocument, boxRect)
{
    Value = "Type your content here..."
};
pdfDocument.Form.Add(placeholderField);

//ExStep:3-
ITaggedContent taggedContent = pdfDocument.TaggedContent;
StructureElement structureRoot = taggedContent.RootElement;
FormElement logicalFormElement = taggedContent.CreateFormElement();
structureRoot.AppendChild(logicalFormElement);
logicalFormElement.Tag(placeholderField);

//ExStep:4-
pdfDocument.Save(resultPath);
}

//ExStep:5-
Console.WriteLine($"PDF with form field saved to '{resultPath}'.");
//ExEnd
