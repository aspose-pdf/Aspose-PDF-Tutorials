//ExSummary: Tutorial ini menunjukkan cara membandingkan dua dokumen PDF dengan Aspose.Pdf untuk .NET sambil mengecualikan semua area anotasi, dengan mengumpulkan kotak pembatas anotasi dari setiap file, mengonfigurasi SideBySideComparisonOptions dengan wilayah tersebut, dan menjalankan perbandingan berdampingan yang menghasilkan PDF diff.
//ExStepSummary:0: Tentukan jalur file untuk PDF utama, PDF sekunder, dan PDF diff output.
//ExStepSummary:1: Periksa bahwa kedua file PDF input ada dan batalkan dengan pesan error jika salah satu tidak ada.
//ExStepSummary:2: Buka kedua file PDF sebagai objek Document (pdfA dan pdfB) dalam blok using sehingga mereka dibuang secara otomatis.
//ExStepSummary:3: Kumpulkan persegi panjang pembatas setiap anotasi di pdfA dan simpan dalam ignoredRectsA.
//ExStepSummary:4: Kumpulkan persegi panjang pembatas setiap anotasi di pdfB dan simpan dalam ignoredRectsB.
//ExStepSummary:5: Buat objek SideBySideComparisonOptions dan tetapkan persegi panjang anotasi yang telah dikumpulkan sebelumnya ke ExcludeAreas1 dan ExcludeAreas2 sehingga wilayah tersebut diabaikan selama perbandingan.
//ExStepSummary:6: Jalankan perbandingan PDF berdampingan menggunakan dua objek Document dan opsi yang dikonfigurasi, menulis hasilnya ke outputPdf.
//ExStepSummary:7: Buang objek Document dan keluar dari blok using.
//ExStepSummary:8: Cetak pesan konfirmasi yang menunjukkan bahwa perbandingan selesai dan di mana PDF diff disimpan.
//ExStep:0-
const string primaryPdf = "sourceA.pdf";
const string secondaryPdf = "sourceB.pdf";
const string outputPdf = "diff_output.pdf";

//ExStep:1-
if (!File.Exists(primaryPdf) || !File.Exists(secondaryPdf))
{
    Console.Error.WriteLine("Required input files are missing.");
    return;
}

//ExStep:2-
using (Document pdfA = new Document(primaryPdf))
using (Document pdfB = new Document(secondaryPdf))
{

//ExStep:3-
// Gather bounding boxes of all annotations in the first PDF
List<Aspose.Pdf.Rectangle> ignoredRectsA = new List<Aspose.Pdf.Rectangle>();
foreach (Page p in pdfA.Pages)
{
    foreach (Annotation a in p.Annotations)
    {
        // Each annotation provides its location via the Rect property
        if (a != null && a.Rect != null)
            ignoredRectsA.Add(a.Rect);
    }
}

//ExStep:4-
// Gather bounding boxes of all annotations in the second PDF
List<Aspose.Pdf.Rectangle> ignoredRectsB = new List<Aspose.Pdf.Rectangle>();
foreach (Page p in pdfB.Pages)
{
    foreach (Annotation a in p.Annotations)
    {
        if (a != null && a.Rect != null)
            ignoredRectsB.Add(a.Rect);
    }
}

//ExStep:5-
// Set up comparison settings to omit the collected regions
SideBySideComparisonOptions compareOpts = new SideBySideComparisonOptions
{
    ExcludeAreas1 = ignoredRectsA.ToArray(),
    ExcludeAreas2 = ignoredRectsB.ToArray()
    // Additional settings may be adjusted here (e.g., ExcludeTables = true)
};

//ExStep:6-
// Execute side‑by‑side comparison and write the outcome
SideBySidePdfComparer.Compare(pdfA, pdfB, outputPdf, compareOpts);

//ExStep:7-
}

//ExStep:8-
Console.WriteLine($"Comparison finished. Output written to '{outputPdf}'.");
//ExEnd
