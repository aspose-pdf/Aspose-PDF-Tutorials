//ExSummary: Tutorial ini menunjukkan cara menggunakan Aspose.Pdf untuk membandingkan dua file PDF yang berisi bidang formulir, secara otomatis mengevaluasi nilai bidang, dan kemudian menghasilkan laporan JSON perbedaan serta PDF visual perbandingan side‑by‑side.
//ExStepSummary:0: Deklarasikan jalur file untuk PDF asli, PDF yang diperbarui, PDF diff side‑by‑side, dan laporan diff JSON.
//ExStepSummary:1: Periksa bahwa kedua file PDF sumber ada dan hentikan program dengan pesan error jika salah satunya tidak ada.
//ExStepSummary:2: Muat PDF asli dan yang diperbarui ke dalam objek `Document` di dalam blok `using` yang deterministik.
//ExStepSummary:3: Buat instance `ComparisonOptions` yang menyimpan pengaturan default untuk perbandingan.
//ExStepSummary:4: Jalankan perbandingan teks halaman demi halaman dari kedua PDF dan simpan operasi diff yang dihasilkan.
//ExStepSummary:5: Hasilkan file JSON yang merinci perbedaan dan tulis pesan konfirmasi ke konsol.
//ExStepSummary:6: Buat opsi perbandingan side‑by‑side, hasilkan PDF visual diff yang menampilkan perubahan, dan catat jalur file output.
//ExStep:0-
const string sourcePdf = "base_document.pdf";
const string updatedPdf = "altered_document.pdf";
const string sideBySidePdf = "diff_visual.pdf";
const string jsonReportPath = "diff_report.json";

//ExStep:1-
if (!File.Exists(sourcePdf) || !File.Exists(updatedPdf))
{
    Console.Error.WriteLine("Required PDF files were not found.");
    return;
}

//ExStep:2-
using (Document firstDoc = new Document(sourcePdf))
using (Document secondDoc = new Document(updatedPdf))
{

//ExStep:3-
ComparisonOptions compareOpts = new ComparisonOptions();

//ExStep:4-
List<List<DiffOperation>> pageDiffs = TextPdfComparer.CompareDocumentsPageByPage(firstDoc, secondDoc, compareOpts);

//ExStep:5-
JsonDiffOutputGenerator jsonWriter = new JsonDiffOutputGenerator();
jsonWriter.GenerateOutput(pageDiffs, jsonReportPath);
Console.WriteLine($"JSON comparison saved to '{jsonReportPath}'.");

//ExStep:6-
SideBySideComparisonOptions sideBySideOpts = new SideBySideComparisonOptions();
SideBySidePdfComparer.Compare(firstDoc, secondDoc, sideBySidePdf, sideBySideOpts);
Console.WriteLine($"Side‑by‑side diff PDF saved to '{sideBySidePdf}'.");
}
