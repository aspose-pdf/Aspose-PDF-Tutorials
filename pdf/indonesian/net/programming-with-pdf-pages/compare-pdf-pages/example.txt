//ExSummary: Tutorial ini menunjukkan cara menggunakan TextPdfComparer dari Aspose.Pdf untuk melakukan perbandingan halaman demi halaman dari dua dokumen PDF dan menghasilkan PDF laporan perbedaan.
//ExStepSummary:0: Tentukan jalur file untuk PDF asli, PDF revisi, dan laporan perbedaan output.
//ExStepSummary:1: Verifikasi bahwa kedua PDF sumber ada dan hentikan dengan pesan error jika salah satunya tidak ada.
//ExStepSummary:2: Buka kedua PDF, buat opsi perbandingan default, jalankan perbandingan halaman demi halaman dengan `TextPdfComparer.CompareDocumentsPageByPage`, dan tampilkan pesan keberhasilan.
//ExStepSummary:3: Tangkap semua pengecualian yang terjadi selama perbandingan dan tulis pesan error ke konsol.
//ExStep:0-
const string sourcePath1 = "original.pdf";
const string sourcePath2 = "revised.pdf";
const string resultPath = "changes_report.pdf";

//ExStep:1-
if (!File.Exists(sourcePath1) || !File.Exists(sourcePath2))
{
    Console.Error.WriteLine("Either of the input PDFs could not be located.");
    return;
}

//ExStep:2-
try
{
    using (Document pdfA = new Document(sourcePath1))
    using (Document pdfB = new Document(sourcePath2))
    {
        ComparisonOptions compareOpts = new ComparisonOptions();
        TextPdfComparer.CompareDocumentsPageByPage(pdfA, pdfB, compareOpts, resultPath);
    }

    Console.WriteLine($"Page‑by‑page diff PDF written to '{resultPath}'.");
}

//ExStep:3-
catch (Exception ex)
{
    Console.Error.WriteLine($"Comparison failed: {ex.Message}");
}
//ExEnd
