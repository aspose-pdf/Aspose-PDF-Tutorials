//ExSummary: Tutorial ini menunjukkan cara membuka PDF yang sudah ada dengan Aspose.Pdf, menerapkan enkripsi AES‑256 tanpa memberikan izin apa pun (yang menonaktifkan pencetakan), menyimpan file yang dilindungi, dan memverifikasi enkripsi menggunakan kata sandi viewer.
//ExStepSummary:0: Tentukan jalur PDF sumber, nama file output yang dilindungi, serta kata sandi viewer dan owner.
//ExStepSummary:1: Periksa apakah file sumber ada dan batalkan dengan pesan error jika tidak ditemukan.
//ExStepSummary:2: Buka PDF asli, tidak mengatur izin apa pun, enkripsi dengan AES‑256 menggunakan kata sandi yang ditentukan, dan simpan file yang dilindungi.
//ExStepSummary:3: Buka kembali PDF yang dienkripsi dengan kata sandi viewer dan tulis flag IsEncrypted‑nya ke console untuk mengonfirmasi perlindungan.
//ExStepSummary:4: Cetak pesan akhir yang menunjukkan bahwa enkripsi selesai dan nama file output.
//ExStep:0-
const string sourceFile = "original.pdf";
const string securedFile = "protected.pdf";
const string viewerPwd = "viewerPwd123";
const string masterPwd = "masterPwd123";

//ExStep:1-
if (!File.Exists(sourceFile))
{
    Console.Error.WriteLine($"Unable to locate file: {sourceFile}");
    return;
}

//ExStep:2-
using (Document pdfDoc = new Document(sourceFile))
{
    Permissions rights = 0;
    pdfDoc.Encrypt(viewerPwd, masterPwd, rights, CryptoAlgorithm.AESx256);
    pdfDoc.Save(securedFile);
}

//ExStep:3-
using (Document checkDoc = new Document(securedFile, viewerPwd))
{
    Console.WriteLine($"Encrypted flag: {checkDoc.IsEncrypted}");
}

//ExStep:4-
Console.WriteLine($"Encryption finished. Output saved as '{securedFile}'.");
//ExEnd
