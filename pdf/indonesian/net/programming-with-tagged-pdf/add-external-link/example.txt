//ExSummary: Tutorial ini menunjukkan cara membuka PDF yang sudah ada dengan Aspose.Pdf, membuat hyperlink eksternal yang ditandai yang mencakup teks tampilan dan judul tooltip, menyisipkannya ke dalam struktur logis dokumen, dan menyimpan file yang telah dimodifikasi.
//ExStepSummary:0: Definisikan jalur untuk PDF sumber dan file hasil.
//ExStepSummary:1: Periksa apakah PDF sumber ada dan batalkan jika tidak dapat ditemukan.
//ExStepSummary:2: Buka dokumen PDF di dalam blok using untuk memastikan pembuangan yang tepat.
//ExStepSummary:3: Dapatkan manager konten bertag untuk dokumen yang dibuka.
//ExStepSummary:4: Setel bahasa dokumen ke English (US) dan beri PDF judul yang diambil dari nama file.
//ExStepSummary:5: Ambil elemen root dari pohon struktur logis tempat elemen baru akan ditambahkan.
//ExStepSummary:6: Buat elemen link, atur teks tampilan, URL target, dan judul tooltip, lalu sisipkan ke dalam struktur dokumen.
//ExStepSummary:7: Simpan PDF yang diperbarui ke file hasil yang ditentukan.
//ExStepSummary:8: Tampilkan pesan konfirmasi yang menunjukkan di mana PDF yang dimodifikasi disimpan.
//ExStep:0-
const string sourceFile = "source.pdf";
const string resultFile = "result_with_external_link.pdf";

//ExStep:1-
if (!File.Exists(sourceFile))
{
    Console.Error.WriteLine($"Unable to locate file: {sourceFile}");
    return;
}

//ExStep:2-
using (Document pdfDoc = new Document(sourceFile))
{

//ExStep:3-
    ITaggedContent taggedContent = pdfDoc.TaggedContent;

//ExStep:4-
    taggedContent.SetLanguage("en-US");
    taggedContent.SetTitle(Path.GetFileNameWithoutExtension(sourceFile));

//ExStep:5-
    StructureElement rootElement = taggedContent.RootElement;

//ExStep:6-
    LinkElement externalLink = taggedContent.CreateLinkElement();
    externalLink.SetText("Open Aspose PDF library");
    externalLink.Hyperlink = new WebHyperlink("https://www.example.com/pdf");
    externalLink.Title = "Example PDF product page";
    rootElement.AppendChild(externalLink);

//ExStep:7-
    pdfDoc.Save(resultFile);
}

//ExStep:8-
Console.WriteLine($"Modified PDF saved as '{resultFile}'.");
