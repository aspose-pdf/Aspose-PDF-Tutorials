//ExSummary: Ce tutoriel montre comment faire pivoter chaque page d'un PDF existant de 180 degrés en utilisant la classe Document d'Aspose.Pdf.
//ExStepSummary:0: Déclarez les chemins d'accès du fichier source et du fichier de destination pour le PDF.
//ExStepSummary:1: Vérifiez que le fichier source existe et interrompez l'exécution avec un message d'erreur s'il n'existe pas.
//ExStepSummary:2: Créez un objet Document pour le PDF source à l'intérieur d'un bloc using afin de gérer les ressources.
//ExStepSummary:3: Parcourez chaque Page dans pdfDoc.Pages et définissez sa propriété Rotate sur Rotation.on180, retournant chaque page à l'envers.
//ExStepSummary:4: Enregistrez le Document modifié dans le fichier de destination, en conservant les rotations.
//ExStepSummary:5: Écrivez un message de confirmation dans la console indiquant que l'opération est terminée.
//ExStepImage:3:images/1.png
//ExStep:0-
        const string sourceFile = "source_document.pdf";
        const string destinationFile = "flipped_document.pdf";

//ExStep:1-
        if (!File.Exists(sourceFile))
        {
            Console.Error.WriteLine($"Unable to locate file: {sourceFile}");
            return;
        }

//ExStep:2-
        // Open the PDF within a using statement to ensure resources are released
        using (Document pdfDoc = new Document(sourceFile))
        {

//ExStep:3-
            // Walk through each page (collection is 1‑based) and apply a 180° rotation
            foreach (Page pdfPage in pdfDoc.Pages)
            {
                // Use the correct enum member with the 'on' prefix
                pdfPage.Rotate = Rotation.on180;
            }

//ExStep:4-
            // Write the altered document to disk
            pdfDoc.Save(destinationFile);
        }

//ExStep:5-
        Console.WriteLine($"All pages rotated and saved as '{destinationFile}'.");
//ExEnd
