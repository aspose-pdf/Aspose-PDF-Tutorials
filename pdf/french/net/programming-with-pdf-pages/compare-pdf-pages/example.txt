//ExSummary: Ce tutoriel montre comment utiliser TextPdfComparer d'Aspose.Pdf pour effectuer une comparaison page par page de deux documents PDF et générer un PDF de rapport de différences.
//ExStepSummary:0: Déclarez les chemins de fichier pour le PDF original, le PDF révisé et le rapport de différences en sortie.
//ExStepSummary:1: Vérifiez que les deux PDF source existent et arrêtez l'exécution avec un message d'erreur si l'un d'eux est manquant.
//ExStepSummary:2: Ouvrez les deux PDF, créez des options de comparaison par défaut, exécutez une différence page par page avec `TextPdfComparer.CompareDocumentsPageByPage`, puis affichez un message de succès.
//ExStepSummary:3: Capturez toute exception levée pendant la comparaison et écrivez un message d'erreur dans la console.
//ExStep:0-
const string sourcePath1 = "original.pdf";
const string sourcePath2 = "revised.pdf";
const string resultPath = "changes_report.pdf";

//ExStep:1-
if (!File.Exists(sourcePath1) || !File.Exists(sourcePath2))
{
    Console.Error.WriteLine("Either of the input PDFs could not be located.");
    return;
}

//ExStep:2-
try
{
    using (Document pdfA = new Document(sourcePath1))
    using (Document pdfB = new Document(sourcePath2))
    {
        ComparisonOptions compareOpts = new ComparisonOptions();
        TextPdfComparer.CompareDocumentsPageByPage(pdfA, pdfB, compareOpts, resultPath);
    }

    Console.WriteLine($"Page‑by‑page diff PDF written to '{resultPath}'.");
}

//ExStep:3-
catch (Exception ex)
{
    Console.Error.WriteLine($"Comparison failed: {ex.Message}");
}
//ExEnd
