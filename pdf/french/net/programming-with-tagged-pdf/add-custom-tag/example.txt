//ExStart
//ExSummary: Ce tutoriel montre comment ajouter une balise personnalisée définie par l'utilisateur à un paragraphe dans un PDF en utilisant les classes Document et ITaggedContent.
//ExStepSummary:0: Définissez le nom du fichier de sortie pour le PDF généré.
//ExStepSummary:1: Créez une nouvelle instance de document PDF vide nommée pdfDoc.
//ExStepSummary:2: Obtenez l'interface ITaggedContent à partir de pdfDoc pour travailler avec les structures PDF balisées.
//ExStepSummary:3: Définissez la langue du document sur English (US) et attribuez un titre aux métadonnées d'accessibilité.
//ExStepSummary:4: Récupérez l'élément racine de l'arbre de structure du PDF.
//ExStepSummary:5: Créez un nouvel élément de paragraphe, attribuez‑lui une balise personnalisée "MyCustomTag", et définissez son texte affiché.
//ExStepSummary:6: Ajoutez le paragraphe personnalisé à l'élément de structure racine, l'insérant ainsi dans la mise en page du document.
//ExStepSummary:7: Enregistrez le PDF construit à l'emplacement de fichier stocké dans resultFile et fermez le périmètre du document.
//ExStepSummary:8: Écrivez un message console confirmant l'emplacement où le PDF a été enregistré.
//ExStepImage:5:images/1.png
//ExStepImage:6:images/2.png
//ExStep:0-
const string resultFile = "custom_tag_demo.pdf";

//ExStep:1-
using (Document pdfDoc = new Document())
{

//ExStep:2-
ITaggedContent tagged = pdfDoc.TaggedContent;

//ExStep:3-
tagged.SetLanguage("en-US");
tagged.SetTitle("Sample PDF Featuring a Custom Paragraph Tag");

//ExStep:4-
StructureElement rootElement = tagged.RootElement;

//ExStep:5-
ParagraphElement customParagraph = tagged.CreateParagraphElement();
customParagraph.SetTag("MyCustomTag");
customParagraph.SetText("The following paragraph demonstrates the use of a user‑defined tag.");

//ExStep:6-
rootElement.AppendChild(customParagraph);

//ExStep:7-
pdfDoc.Save(resultFile);
}

//ExStep:8-
Console.WriteLine($"PDF saved to '{resultFile}'.");
//ExEnd
