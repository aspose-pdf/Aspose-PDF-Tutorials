//ExStart
//ExSummary: This tutorial demonstrates how to open an existing PDF with Aspose.Pdf, apply AES‑256 encryption without granting any permissions (which disables printing), save the protected file, and verify the encryption using the viewer password.
//ExStepSummary:0: Define the source PDF path, the output protected file name, and the viewer and owner passwords.
//ExStepSummary:1: Check that the source file exists and abort with an error message if it is missing.
//ExStepSummary:2: Open the original PDF, set no permissions, encrypt it with AES‑256 using the specified passwords, and save the protected file.
//ExStepSummary:3: Re‑open the encrypted PDF with the viewer password and write its IsEncrypted flag to the console to confirm protection.
//ExStepSummary:4: Print a final message indicating that encryption is complete and the output file name.
//ExStep:0-
const string sourceFile = "original.pdf";
const string securedFile = "protected.pdf";
const string viewerPwd = "viewerPwd123";
const string masterPwd = "masterPwd123";

//ExStep:1-
if (!File.Exists(sourceFile))
{
    Console.Error.WriteLine($"Unable to locate file: {sourceFile}");
    return;
}

//ExStep:2-
using (Document pdfDoc = new Document(sourceFile))
{
    Permissions rights = 0;
    pdfDoc.Encrypt(viewerPwd, masterPwd, rights, CryptoAlgorithm.AESx256);
    pdfDoc.Save(securedFile);
}

//ExStep:3-
using (Document checkDoc = new Document(securedFile, viewerPwd))
{
    Console.WriteLine($"Encrypted flag: {checkDoc.IsEncrypted}");
}

//ExStep:4-
Console.WriteLine($"Encryption finished. Output saved as '{securedFile}'.");
//ExEnd
