//ExSummary: This tutorial demonstrates how to open an existing PDF with Aspose.Pdf, create a tagged external hyperlink that includes display text and a tooltip title, insert it into the document's logical structure, and save the modified file.
//ExStepSummary:0: Define the paths for the source PDF and the result file.
//ExStepSummary:1: Check that the source PDF exists and abort if it cannot be found.
//ExStepSummary:2: Open the PDF document inside a using block to ensure proper disposal.
//ExStepSummary:3: Obtain the tagged‑content manager for the opened document.
//ExStepSummary:4: Set the document language to English (US) and give the PDF a title derived from the file name.
//ExStepSummary:5: Retrieve the root element of the logical structure tree to which new elements will be added.
//ExStepSummary:6: Create a link element, set its displayed text, target URL, and tooltip title, then insert it into the document’s structure.
//ExStepSummary:7: Save the updated PDF to the specified result file.
//ExStepSummary:8: Output a confirmation message indicating where the modified PDF was saved.
//ExStep:0-
const string sourceFile = "source.pdf";
const string resultFile = "result_with_external_link.pdf";

//ExStep:1-
if (!File.Exists(sourceFile))
{
    Console.Error.WriteLine($"Unable to locate file: {sourceFile}");
    return;
}

//ExStep:2-
using (Document pdfDoc = new Document(sourceFile))
{

//ExStep:3-
    ITaggedContent taggedContent = pdfDoc.TaggedContent;

//ExStep:4-
    taggedContent.SetLanguage("en-US");
    taggedContent.SetTitle(Path.GetFileNameWithoutExtension(sourceFile));

//ExStep:5-
    StructureElement rootElement = taggedContent.RootElement;

//ExStep:6-
    LinkElement externalLink = taggedContent.CreateLinkElement();
    externalLink.SetText("Open Aspose PDF library");
    externalLink.Hyperlink = new WebHyperlink("https://www.example.com/pdf");
    externalLink.Title = "Example PDF product page";
    rootElement.AppendChild(externalLink);

//ExStep:7-
    pdfDoc.Save(resultFile);
}

//ExStep:8-
Console.WriteLine($"Modified PDF saved as '{resultFile}'.");
//ExEnd
