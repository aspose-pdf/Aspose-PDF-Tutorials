//ExSummary: यह ट्यूटोरियल दिखाता है कि Aspose.Pdf Document क्लास का उपयोग करके मौजूदा PDF के प्रत्येक पृष्ठ को 180 डिग्री घुमाया जाए।
//ExStepSummary:0: PDF के स्रोत और गंतव्य फ़ाइल पथ घोषित करें।
//ExStepSummary:1: जाँचें कि स्रोत फ़ाइल मौजूद है या नहीं और यदि नहीं है तो त्रुटि संदेश के साथ प्रक्रिया रोक दें।
//ExStepSummary:2: संसाधनों को प्रबंधित करने के लिए एक using ब्लॉक के भीतर स्रोत PDF के लिए Document ऑब्जेक्ट बनाएं।
//ExStepSummary:3: pdfDoc.Pages में प्रत्येक Page पर इटररेट करें और उसकी Rotate प्रॉपर्टी को Rotation.on180 सेट करें, जिससे प्रत्येक पृष्ठ उल्टा हो जाएगा।
//ExStepSummary:4: परिवर्तित Document को गंतव्य फ़ाइल में सहेजें, जिससे घुमाव स्थायी हो जाएँ।
//ExStepSummary:5: ऑपरेशन पूर्ण होने का संकेत देने के लिए कंसोल में एक पुष्टि संदेश लिखें।
//ExStepImage:3:images/1.png
//ExStep:0-
        const string sourceFile = "source_document.pdf";
        const string destinationFile = "flipped_document.pdf";

//ExStep:1-
        if (!File.Exists(sourceFile))
        {
            Console.Error.WriteLine($"Unable to locate file: {sourceFile}");
            return;
        }

//ExStep:2-
        // Open the PDF within a using statement to ensure resources are released
        using (Document pdfDoc = new Document(sourceFile))
        {

//ExStep:3-
            // Walk through each page (collection is 1‑based) and apply a 180° rotation
            foreach (Page pdfPage in pdfDoc.Pages)
            {
                // Use the correct enum member with the 'on' prefix
                pdfPage.Rotate = Rotation.on180;
            }

//ExStep:4-
            // Write the altered document to disk
            pdfDoc.Save(destinationFile);
        }

//ExStep:5-
        Console.WriteLine($"All pages rotated and saved as '{destinationFile}'.");
//ExEnd
