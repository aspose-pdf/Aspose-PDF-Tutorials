//ExSummary: Tento tutoriál ukazuje, jak otevřít existující PDF pomocí Aspose.Pdf, vytvořit označený externí hypertextový odkaz, který obsahuje zobrazovaný text a titulek tooltipu, vložit jej do logické struktury dokumentu a uložit upravený soubor.
//ExStepSummary:0: Definujte cesty pro zdrojové PDF a soubor výsledku.
//ExStepSummary:1: Ověřte, že zdrojové PDF existuje, a pokud jej nelze najít, přerušte operaci.
//ExStepSummary:2: Otevřete PDF dokument v bloku using, aby byl zajištěn řádný uvolnění prostředků.
//ExStepSummary:3: Získejte správce označeného obsahu (tagged‑content manager) pro otevřený dokument.
//ExStepSummary:4: Nastavte jazyk dokumentu na angličtinu (US) a dejte PDF titulek odvozený od názvu souboru.
//ExStepSummary:5: Získejte kořenový prvek stromu logické struktury, do kterého budou přidány nové prvky.
//ExStepSummary:6: Vytvořte prvek odkazu, nastavte jeho zobrazovaný text, cílovou URL a titulek tooltipu a poté jej vložte do struktury dokumentu.
//ExStepSummary:7: Uložte aktualizované PDF do určeného výstupního souboru.
//ExStepSummary:8: Vypište potvrzovací zprávu, která uvádí, kam byl upravený PDF soubor uložen.
//ExStep:0-
const string sourceFile = "source.pdf";
const string resultFile = "result_with_external_link.pdf";

//ExStep:1-
if (!File.Exists(sourceFile))
{
    Console.Error.WriteLine($"Unable to locate file: {sourceFile}");
    return;
}

//ExStep:2-
using (Document pdfDoc = new Document(sourceFile))
{

//ExStep:3-
    ITaggedContent taggedContent = pdfDoc.TaggedContent;

//ExStep:4-
    taggedContent.SetLanguage("en-US");
    taggedContent.SetTitle(Path.GetFileNameWithoutExtension(sourceFile));

//ExStep:5-
    StructureElement rootElement = taggedContent.RootElement;

//ExStep:6-
    LinkElement externalLink = taggedContent.CreateLinkElement();
    externalLink.SetText("Open Aspose PDF library");
    externalLink.Hyperlink = new WebHyperlink("https://www.example.com/pdf");
    externalLink.Title = "Example PDF product page";
    rootElement.AppendChild(externalLink);

//ExStep:7-
    pdfDoc.Save(resultFile);
}

//ExStep:8-
Console.WriteLine($"Modified PDF saved as '{resultFile}'.");
//ExEnd
