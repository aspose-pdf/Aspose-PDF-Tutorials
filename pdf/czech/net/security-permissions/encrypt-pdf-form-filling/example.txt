//ExSummary: Tento tutoriál ukazuje, jak otevřít existující PDF pomocí Aspose.Pdf, omezit oprávnění tak, aby bylo povoleno pouze vyplňování formulářů, zašifrovat jej pomocí hesel pro prohlížeče a vlastníka s využitím AES‑256 a uložit chráněný soubor.
//ExStepSummary:0: Definujte cesty ke zdrojovému a cílovému souboru a hesla pro prohlížeče a vlastníka.
//ExStepSummary:1: Zkontrolujte, zda zdrojové PDF existuje, a pokud ne, ukončete s chybovou zprávou.
//ExStepSummary:2: Začněte blokem try, abyste zachytili případné chyby, které mohou během zpracování nastat.
//ExStepSummary:3: Otevřete existující PDF soubor vytvořením objektu Document s názvem pdfDoc.
//ExStepSummary:4: Nastavte oprávnění tak, aby bylo povoleno pouze vyplňování formulářů, zašifrujte dokument pomocí hesel pro prohlížeče a vlastníka s využitím AES‑256 a uložte chráněné PDF.
//ExStepSummary:5: Napište potvrzovací zprávu do konzole a uzavřete blok try.
//ExStepSummary:6: Zachyťte všechny výjimky a vypište chybovou zprávu do konzole.
//ExStep:0-
        const string sourceFile = "sample.pdf";
        const string destFile = "protected.pdf";
        const string viewerPass = "viewerPass";
        const string adminPass = "adminPass";

//ExStep:1-
        if (!File.Exists(sourceFile))
        {
            Console.Error.WriteLine($"Unable to locate file: {sourceFile}");
            return;
        }

//ExStep:2-
        try
        {

//ExStep:3-
            using (Document pdfDoc = new Document(sourceFile))

//ExStep:4-
            {
                Permissions allowed = Permissions.FillForm;
                pdfDoc.Encrypt(viewerPass, adminPass, allowed, CryptoAlgorithm.AESx256);
                pdfDoc.Save(destFile);
            }

//ExStep:5-
            Console.WriteLine($"Secure PDF written to '{destFile}'.");
        }

//ExStep:6-
        catch (Exception ex)
        {
            Console.Error.WriteLine($"Exception occurred: {ex.Message}");
        }
//ExEnd
