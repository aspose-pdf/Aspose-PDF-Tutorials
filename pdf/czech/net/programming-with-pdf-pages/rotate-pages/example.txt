//ExSummary: Tento tutoriál ukazuje, jak otočit každou stránku existujícího PDF o 180 stupňů pomocí třídy Aspose.Pdf Document.
//ExStepSummary:0: Deklarujte cesty ke zdrojovému a cílovému souboru PDF.
//ExStepSummary:1: Zkontrolujte, zda zdrojový soubor existuje, a v případě, že ne, ukončete program s chybovou zprávou.
//ExStepSummary:2: Vytvořte objekt Document pro zdrojové PDF uvnitř bloku using, aby se spravovaly prostředky.
//ExStepSummary:3: Procházejte každou stránku v pdfDoc.Pages a nastavte její vlastnost Rotate na Rotation.on180, čímž otočíte každou stránku vzhůru nohama.
//ExStepSummary:4: Uložte upravený Document do cílového souboru, čímž trvale zachováte otočení.
//ExStepSummary:5: Vypište potvrzovací zprávu do konzole, která naznačuje, že operace byla dokončena.
//ExStepImage:3:images/1.png
//ExStep:0-
        const string sourceFile = "source_document.pdf";
        const string destinationFile = "flipped_document.pdf";

//ExStep:1-
        if (!File.Exists(sourceFile))
        {
            Console.Error.WriteLine($"Unable to locate file: {sourceFile}");
            return;
        }

//ExStep:2-
        // Open the PDF within a using statement to ensure resources are released
        using (Document pdfDoc = new Document(sourceFile))
        {

//ExStep:3-
            // Walk through each page (collection is 1‑based) and apply a 180° rotation
            foreach (Page pdfPage in pdfDoc.Pages)
            {
                // Use the correct enum member with the 'on' prefix
                pdfPage.Rotate = Rotation.on180;
            }

//ExStep:4-
            // Write the altered document to disk
            pdfDoc.Save(destinationFile);
        }

//ExStep:5-
        Console.WriteLine($"All pages rotated and saved as '{destinationFile}'.");
//ExEnd
