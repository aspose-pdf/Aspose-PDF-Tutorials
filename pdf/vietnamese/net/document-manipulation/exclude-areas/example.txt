//ExSummary: Hướng dẫn này cho thấy cách sử dụng Aspose.Pdf để so sánh hai tệp PDF cạnh nhau trong khi bỏ qua các vùng hình chữ nhật cụ thể trong mỗi tài liệu, tạo ra một tệp PDF diff chỉ làm nổi bật các thay đổi liên quan.
//ExStepSummary:0: Xác định đường dẫn tệp cho hai PDF nguồn và PDF diff đầu ra.
//ExStepSummary:1: Kiểm tra xem cả hai tệp PDF nguồn có tồn tại hay không và hủy bỏ với thông báo lỗi nếu thiếu bất kỳ tệp nào.
//ExStepSummary:2: Tạo một mảng chứa vùng hình chữ nhật cần bỏ qua trong PDF đầu tiên.
//ExStepSummary:3: Tạo một mảng chứa vùng hình chữ nhật cần bỏ qua trong PDF thứ hai.
//ExStepSummary:4: Cấu hình việc so sánh cạnh nhau, gán các mảng vùng bỏ qua vào các thuộc tính tương ứng.
//ExStepSummary:5: Tải cả hai PDF và thực hiện so sánh cạnh nhau, tạo ra tệp diff tại đường dẫn đã chỉ định.
//ExStepSummary:6: In ra thông báo xác nhận cho biết việc so sánh đã hoàn thành và PDF diff đã được lưu ở đâu.
//ExStep:0-
const string sourcePdfA = "originalA.pdf";
const string sourcePdfB = "originalB.pdf";
const string diffPdfPath = "diff_output.pdf";

//ExStep:1-
if (!File.Exists(sourcePdfA) || !File.Exists(sourcePdfB))
{
    Console.Error.WriteLine("Required PDF files are missing.");
    return;
}

//ExStep:2-
Aspose.Pdf.Rectangle[] zonesToIgnoreA = new Aspose.Pdf.Rectangle[]
{
    new Aspose.Pdf.Rectangle(100, 500, 300, 600) // sample region in the first document
};

//ExStep:3-
Aspose.Pdf.Rectangle[] zonesToIgnoreB = new Aspose.Pdf.Rectangle[]
{
    new Aspose.Pdf.Rectangle(50, 400, 250, 500) // sample region in the second document
};

//ExStep:4-
var comparisonSettings = new SideBySideComparisonOptions
{
    ExcludeAreas1 = zonesToIgnoreA,
    ExcludeAreas2 = zonesToIgnoreB
    // Other settings may be adjusted here, e.g., ExcludeTables = true;
};

//ExStep:5-
using (Document pdfA = new Document(sourcePdfA))
using (Document pdfB = new Document(sourcePdfB))
{
    SideBySidePdfComparer.Compare(pdfA, pdfB, diffPdfPath, comparisonSettings);
}

//ExStep:6-
Console.WriteLine($"Comparison finished. Output written to '{diffPdfPath}'.");
//ExEnd
