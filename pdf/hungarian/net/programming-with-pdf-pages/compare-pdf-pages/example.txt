//ExSummary: Ez a bemutató azt mutatja be, hogyan kell használni az Aspose.Pdf TextPdfComparer osztályát két PDF dokumentum oldalankénti összehasonlításához és egy diff jelentés PDF generálásához.
//ExStepSummary:0: Határozza meg az eredeti PDF, a módosított PDF és a kimeneti diff jelentés fájlútvonalait.
//ExStepSummary:1: Ellenőrizze, hogy mindkét forrás‑PDF létezik-e, és ha valamelyik hiányzik, hibajelzéssel álljon le.
//ExStepSummary:2: Nyissa meg mindkét PDF‑et, hozza létre az alapértelmezett összehasonlítási beállításokat, hajtson végre oldalankénti diffet a `TextPdfComparer.CompareDocumentsPageByPage` metódussal, és jelenítsen meg egy sikerüzenetet.
//ExStepSummary:3: Fogja el a összehasonlítás során keletkező bármely kivételt, és írjon hibajelzést a konzolra.
//ExStep:0-
const string sourcePath1 = "original.pdf";
const string sourcePath2 = "revised.pdf";
const string resultPath = "changes_report.pdf";

//ExStep:1-
if (!File.Exists(sourcePath1) || !File.Exists(sourcePath2))
{
    Console.Error.WriteLine("Either of the input PDFs could not be located.");
    return;
}

//ExStep:2-
try
{
    using (Document pdfA = new Document(sourcePath1))
    using (Document pdfB = new Document(sourcePath2))
    {
        ComparisonOptions compareOpts = new ComparisonOptions();
        TextPdfComparer.CompareDocumentsPageByPage(pdfA, pdfB, compareOpts, resultPath);
    }

    Console.WriteLine($"Page‑by‑page diff PDF written to '{resultPath}'.");
}

//ExStep:3-
catch (Exception ex)
{
    Console.Error.WriteLine($"Comparison failed: {ex.Message}");
}
//ExEnd
