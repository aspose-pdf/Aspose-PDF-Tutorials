//ExSummary: Ez az útmutató bemutatja, hogyan lehet két PDF dokumentumot összehasonlítani az Aspose.Pdf for .NET segítségével, miközben kizárjuk az összes annotációs területet, az annotációk körülhatároló dobozainak összegyűjtésével minden fájlból, a SideBySideComparisonOptions konfigurálásával ezekkel a régiókkal, és egy oldal‑oldali összehasonlítás futtatásával, amely diff PDF-et eredményez.
//ExStepSummary:0: Határozza meg a fájlútvonalakat az elsődleges PDF, a másodlagos PDF és a kimeneti diff PDF számára.
//ExStepSummary:1: Ellenőrizze, hogy mindkét bemeneti PDF fájl létezik-e, és ha valamelyik hiányzik, abortáljon egy hibaüzenettel.
//ExStepSummary:2: Nyissa meg a két PDF fájlt Document objektumként (pdfA és pdfB) egy using blokkban, hogy azok automatikusan felszabaduljanak.
//ExStepSummary:3: Gyűjtse össze minden annotáció körülhatároló téglalapját a pdfA-ban, és tárolja őket az ignoredRectsA-ban.
//ExStepSummary:4: Gyűjtse össze minden annotáció körülhatároló téglalapját a pdfB-ban, és tárolja őket az ignoredRectsB-ban.
//ExStepSummary:5: Hozzon létre egy SideBySideComparisonOptions objektumot, és rendelje hozzá a korábban összegyűjtött annotációs téglalapokat az ExcludeAreas1 és ExcludeAreas2 tulajdonságokhoz, hogy ezek a területek figyelmen kívül legyenek hagyva az összehasonlítás során.
//ExStepSummary:6: Futtassa az oldal‑oldali PDF összehasonlítást a két Document objektummal és a beállított opciókkal, az eredményt az outputPdf-be írva.
//ExStepSummary:7: Szabadítsa fel a Document objektumokat, és lépjen ki a using blokkból.
//ExStepSummary:8: Írjon ki egy megerősítő üzenetet, amely jelzi, hogy az összehasonlítás befejeződött, és megmutatja, hol lett elmentve a diff PDF.
//ExStep:0-
const string primaryPdf = "sourceA.pdf";
const string secondaryPdf = "sourceB.pdf";
const string outputPdf = "diff_output.pdf";

//ExStep:1-
if (!File.Exists(primaryPdf) || !File.Exists(secondaryPdf))
{
    Console.Error.WriteLine("Required input files are missing.");
    return;
}

//ExStep:2-
using (Document pdfA = new Document(primaryPdf))
using (Document pdfB = new Document(secondaryPdf))
{

//ExStep:3-
// Gather bounding boxes of all annotations in the first PDF
List<Aspose.Pdf.Rectangle> ignoredRectsA = new List<Aspose.Pdf.Rectangle>();
foreach (Page p in pdfA.Pages)
{
    foreach (Annotation a in p.Annotations)
    {
        // Each annotation provides its location via the Rect property
        if (a != null && a.Rect != null)
            ignoredRectsA.Add(a.Rect);
    }
}

//ExStep:4-
// Gather bounding boxes of all annotations in the second PDF
List<Aspose.Pdf.Rectangle> ignoredRectsB = new List<Aspose.Pdf.Rectangle>();
foreach (Page p in pdfB.Pages)
{
    foreach (Annotation a in p.Annotations)
    {
        if (a != null && a.Rect != null)
            ignoredRectsB.Add(a.Rect);
    }
}

//ExStep:5-
// Set up comparison settings to omit the collected regions
SideBySideComparisonOptions compareOpts = new SideBySideComparisonOptions
{
    ExcludeAreas1 = ignoredRectsA.ToArray(),
    ExcludeAreas2 = ignoredRectsB.ToArray()
    // Additional settings may be adjusted here (e.g., ExcludeTables = true)
};

//ExStep:6-
// Execute side‑by‑side comparison and write the outcome
SideBySidePdfComparer.Compare(pdfA, pdfB, outputPdf, compareOpts);

//ExStep:7-
}

//ExStep:8-
Console.WriteLine($"Comparison finished. Output written to '{outputPdf}'.");
//ExEnd
