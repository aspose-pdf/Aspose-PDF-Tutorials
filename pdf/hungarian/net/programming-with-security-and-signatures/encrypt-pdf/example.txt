//ExSummary: Ez a bemutató azt mutatja be, hogyan lehet betölteni egy meglévő PDF-et az Aspose.Pdf segítségével, korlátozni a jogosultságait, titkosítani azt egy felhasználói (kliens) és egy tulajdonosi (admin) jelszóval az AES‑256 algoritmus használatával, és elmenteni a védett dokumentumot.
//ExStepSummary:0: Határozza meg a forrás- és célfájl útvonalakat, valamint a felhasználói (kliens) és tulajdonosi (admin) jelszavakat.
//ExStepSummary:1: Ellenőrizze, hogy a forrás PDF létezik-e, és ha nem, álljon le egy hibaüzenettel.
//ExStepSummary:2: Nyissa meg a PDF-et, engedélyezze csak a tartalom módosítását és kinyerését, titkosítsa a két jelszóval az AES‑256 használatával, mentse a védett fájlt, és jelenítsen meg egy sikerüzenetet.
//ExStepSummary:3: Fogja el a felmerülő hibákat, és írja a kivétel üzenetét a hiba konzolra.
//ExStep:0-
const string sourceFile = "original.pdf";
const string destinationFile = "protected.pdf";
const string clientPwd = "clientPwd";
const string adminPwd = "adminPwd";

//ExStep:1-
if (!File.Exists(sourceFile))
{
    Console.Error.WriteLine($"Unable to locate file: {sourceFile}");
    return;
}

//ExStep:2-
try
{
    using (Document pdfDoc = new Document(sourceFile))
    {
        Permissions allowedPerms = Permissions.ModifyContent | Permissions.ExtractContent;

        pdfDoc.Encrypt(clientPwd, adminPwd, allowedPerms, CryptoAlgorithm.AESx256);

        pdfDoc.Save(destinationFile);
    }

    Console.WriteLine($"Secured PDF written to '{destinationFile}'.");
}

//ExStep:3-
catch (Exception ex)
{
    Console.Error.WriteLine($"Failure: {ex.Message}");
}
//ExEnd
