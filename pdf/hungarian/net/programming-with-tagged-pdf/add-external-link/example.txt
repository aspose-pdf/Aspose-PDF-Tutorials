//ExSummary: Ez az útmutató bemutatja, hogyan nyithat meg egy meglévő PDF-et az Aspose.Pdf segítségével, hogyan hozhat létre egy címkézett külső hiperhivatkozást, amely tartalmaz megjelenítendő szöveget és egy tooltip címet, hogyan illesztheti be a dokumentum logikai struktúrájába, és hogyan mentheti el a módosított fájlt.
//ExStepSummary:0: Határozza meg a forrás PDF és a kimeneti fájl útvonalait.
//ExStepSummary:1: Ellenőrizze, hogy a forrás PDF létezik-e, és szakítsa meg a folyamatot, ha nem található.
//ExStepSummary:2: Nyissa meg a PDF dokumentumot egy using blokkban, hogy biztosítsa a megfelelő erőforrás-felszabadítást.
//ExStepSummary:3: Szerezze meg a címkézett tartalomkezelőt a megnyitott dokumentumhoz.
//ExStepSummary:4: Állítsa be a dokumentum nyelvét English (US) értékre, és adjon a PDF-nek egy, a fájlnévből származó címet.
//ExStepSummary:5: Hozza vissza a logikai struktúrafa gyökérelemét, amelyhez az új elemeket hozzá fogják adni.
//ExStepSummary:6: Hozzon létre egy linkelemet, állítsa be a megjelenített szöveget, a cél URL-t és a tooltip címet, majd illessze be a dokumentum struktúrájába.
//ExStepSummary:7: Mentse el a frissített PDF-et a megadott kimeneti fájlba.
//ExStepSummary:8: Írjon ki egy megerősítő üzenetet, amely jelzi, hová lett mentve a módosított PDF.
//ExStep:0-
const string sourceFile = "source.pdf";
const string resultFile = "result_with_external_link.pdf";

//ExStep:1-
if (!File.Exists(sourceFile))
{
    Console.Error.WriteLine($"Unable to locate file: {sourceFile}");
    return;
}

//ExStep:2-
using (Document pdfDoc = new Document(sourceFile))
{

//ExStep:3-
    ITaggedContent taggedContent = pdfDoc.TaggedContent;

//ExStep:4-
    taggedContent.SetLanguage("en-US");
    taggedContent.SetTitle(Path.GetFileNameWithoutExtension(sourceFile));

//ExStep:5-
    StructureElement rootElement = taggedContent.RootElement;

//ExStep:6-
    LinkElement externalLink = taggedContent.CreateLinkElement();
    externalLink.SetText("Open Aspose PDF library");
    externalLink.Hyperlink = new WebHyperlink("https://www.example.com/pdf");
    externalLink.Title = "Example PDF product page";
    rootElement.AppendChild(externalLink);

//ExStep:7-
    pdfDoc.Save(resultFile);
}

//ExStep:8-
Console.WriteLine($"Modified PDF saved as '{resultFile}'.");
//ExEnd
