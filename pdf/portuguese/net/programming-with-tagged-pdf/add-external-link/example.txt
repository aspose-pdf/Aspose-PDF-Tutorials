//ExSummary: Este tutorial demonstra como abrir um PDF existente com Aspose.Pdf, criar um hyperlink externo marcado que inclui texto de exibição e um título de tooltip, inseri‑lo na estrutura lógica do documento e salvar o arquivo modificado.
//ExStepSummary:0: Defina os caminhos para o PDF de origem e o arquivo de resultado.
//ExStepSummary:1: Verifique se o PDF de origem existe e interrompa a execução se não for encontrado.
//ExStepSummary:2: Abra o documento PDF dentro de um bloco using para garantir a liberação adequada.
//ExStepSummary:3: Obtenha o gerenciador de conteúdo marcado (tagged‑content) para o documento aberto.
//ExStepSummary:4: Defina o idioma do documento como English (US) e atribua ao PDF um título derivado do nome do arquivo.
//ExStepSummary:5: Recupere o elemento raiz da árvore de estrutura lógica ao qual novos elementos serão adicionados.
//ExStepSummary:6: Crie um elemento de link, defina seu texto exibido, URL de destino e título da tooltip, e então insira‑o na estrutura do documento.
//ExStepSummary:7: Salve o PDF atualizado no arquivo de resultado especificado.
//ExStepSummary:8: Exiba uma mensagem de confirmação indicando onde o PDF modificado foi salvo.
//ExStep:0-
const string sourceFile = "source.pdf";
const string resultFile = "result_with_external_link.pdf";

//ExStep:1-
if (!File.Exists(sourceFile))
{
    Console.Error.WriteLine($"Unable to locate file: {sourceFile}");
    return;
}

//ExStep:2-
using (Document pdfDoc = new Document(sourceFile))
{

//ExStep:3-
    ITaggedContent taggedContent = pdfDoc.TaggedContent;

//ExStep:4-
    taggedContent.SetLanguage("en-US");
    taggedContent.SetTitle(Path.GetFileNameWithoutExtension(sourceFile));

//ExStep:5-
    StructureElement rootElement = taggedContent.RootElement;

//ExStep:6-
    LinkElement externalLink = taggedContent.CreateLinkElement();
    externalLink.SetText("Open Aspose PDF library");
    externalLink.Hyperlink = new WebHyperlink("https://www.example.com/pdf");
    externalLink.Title = "Example PDF product page";
    rootElement.AppendChild(externalLink);

//ExStep:7-
    pdfDoc.Save(resultFile);
}

//ExStep:8-
Console.WriteLine($"Modified PDF saved as '{resultFile}'.");
//ExEnd
