//ExStart
//ExSummary: บทแนะนำนี้แสดงวิธีการเปิดไฟล์ PDF ที่มีอยู่ด้วย Aspose.Pdf, สร้างลิงก์ภายนอกที่มีแท็กซึ่งรวมข้อความที่แสดงและชื่อเครื่องมือ (tooltip), แทรกลงในโครงสร้างเชิงตรรกะของเอกสาร, และบันทึกไฟล์ที่แก้ไขแล้ว.
//ExStepSummary:0: กำหนดเส้นทางสำหรับ PDF ต้นฉบับและไฟล์ผลลัพธ์.
//ExStepSummary:1: ตรวจสอบว่า PDF ต้นฉบับมีอยู่และยกเลิกการทำงานหากไม่พบ.
//ExStepSummary:2: เปิดเอกสาร PDF ภายในบล็อก using เพื่อให้แน่ใจว่ามีการจัดการทรัพยากรอย่างเหมาะสม.
//ExStepSummary:3: รับตัวจัดการ tagged‑content สำหรับเอกสารที่เปิดอยู่.
//ExStepSummary:4: ตั้งค่าภาษาเอกสารเป็น English (US) และกำหนดชื่อเรื่องให้กับ PDF โดยอ้างอิงจากชื่อไฟล์.
//ExStepSummary:5: ดึงเอา root element ของโครงสร้างเชิงตรรกะ (logical structure tree) ที่จะเพิ่มองค์ประกอบใหม่เข้าไป.
//ExStepSummary:6: สร้าง LinkElement, ตั้งค่าข้อความที่แสดง, URL ปลายทาง, และชื่อ tooltip, จากนั้นแทรกลงในโครงสร้างของเอกสาร.
//ExStepSummary:7: บันทึก PDF ที่อัปเดตไปยังไฟล์ผลลัพธ์ที่ระบุ.
//ExStepSummary:8: แสดงข้อความยืนยันที่ระบุตำแหน่งที่บันทึก PDF ที่แก้ไขแล้ว.
//ExStep:0-
const string sourceFile = "source.pdf";
const string resultFile = "result_with_external_link.pdf";

//ExStep:1-
if (!File.Exists(sourceFile))
{
    Console.Error.WriteLine($"Unable to locate file: {sourceFile}");
    return;
}

//ExStep:2-
using (Document pdfDoc = new Document(sourceFile))
{

//ExStep:3-
    ITaggedContent taggedContent = pdfDoc.TaggedContent;

//ExStep:4-
    taggedContent.SetLanguage("en-US");
    taggedContent.SetTitle(Path.GetFileNameWithoutExtension(sourceFile));

//ExStep:5-
    StructureElement rootElement = taggedContent.RootElement;

//ExStep:6-
    LinkElement externalLink = taggedContent.CreateLinkElement();
    externalLink.SetText("Open Aspose PDF library");
    externalLink.Hyperlink = new WebHyperlink("https://www.example.com/pdf");
    externalLink.Title = "Example PDF product page";
    rootElement.AppendChild(externalLink);

//ExStep:7-
    pdfDoc.Save(resultFile);
}

//ExStep:8-
Console.WriteLine($"Modified PDF saved as '{resultFile}'.");
//ExEnd
