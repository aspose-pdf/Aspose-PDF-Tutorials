//ExSummary: 本教程演示如何使用 Aspose.Pdf 打开现有 PDF，限制权限仅允许填写表单，使用 AES‑256 通过查看者密码和所有者密码进行加密，并保存受保护的文件。
//ExStepSummary:0: 定义源文件和目标文件的路径以及查看者密码和所有者密码。
//ExStepSummary:1: 检查源 PDF 是否存在，如果不存在则以错误信息中止。
//ExStepSummary:2: 启动 try 块以处理处理过程中可能出现的任何错误。
//ExStepSummary:3: 通过创建名为 pdfDoc 的 Document 对象来打开现有 PDF 文件。
//ExStepSummary:4: 设置仅允许填写表单的权限，使用 AES‑256 通过查看者密码和所有者密码加密文档，并保存受保护的 PDF。
//ExStepSummary:5: 在控制台写入确认信息并关闭 try 块。
//ExStepSummary:6: 捕获任何异常并将错误信息输出到控制台。
//ExStep:0-
        const string sourceFile = "sample.pdf";
        const string destFile = "protected.pdf";
        const string viewerPass = "viewerPass";
        const string adminPass = "adminPass";

//ExStep:1-
        if (!File.Exists(sourceFile))
        {
            Console.Error.WriteLine($"Unable to locate file: {sourceFile}");
            return;
        }

//ExStep:2-
        try
        {

//ExStep:3-
            using (Document pdfDoc = new Document(sourceFile))

//ExStep:4-
            {
                Permissions allowed = Permissions.FillForm;
                pdfDoc.Encrypt(viewerPass, adminPass, allowed, CryptoAlgorithm.AESx256);
                pdfDoc.Save(destFile);
            }

//ExStep:5-
            Console.WriteLine($"Secure PDF written to '{destFile}'.");
        }

//ExStep:6-
        catch (Exception ex)
        {
            Console.Error.WriteLine($"Exception occurred: {ex.Message}");
        }
//ExEnd
